<script type="module">
(() => {
  const host = document.querySelector('[data-sec="box"] [data-box3d]');
  if (!host || !host.dataset.src) return;
  const src = host.dataset.src;

  const YAW_LIMIT   = 0.62;
  const SWAY_SPEED  = 0.00042;
  const RETURN_EASE = 0.055;

  let booted = false;

  const boot = async () => {
    if (booted) return; booted = true;
    let THREE, GLTFLoader, RoomEnvironment;
    try {
      THREE = await import('three');
      ({ GLTFLoader } = await import('three/addons/loaders/GLTFLoader.js'));
      try {
        ({ RoomEnvironment } = await import('three/addons/environments/RoomEnvironment.js'));
      } catch (e) { RoomEnvironment = null; }
    } catch (e) { return; }

    const canvas   = host.querySelector('[data-box-canvas]');
    const fallback = host.querySelector('[data-box-fallback]');
    const hint     = host.querySelector('[data-box-hint]');

    const renderer = new THREE.WebGLRenderer({ canvas, antialias:true, alpha:true });
    renderer.setPixelRatio(Math.min(devicePixelRatio, 2));

    /* ---- COLOUR PIPELINE — this is what was wrong ---- */
    if ('outputColorSpace' in renderer && THREE.SRGBColorSpace) {
      renderer.outputColorSpace = THREE.SRGBColorSpace;
    } else if ('outputEncoding' in renderer && THREE.sRGBEncoding) {
      renderer.outputEncoding = THREE.sRGBEncoding;      // older three
    }
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.15;

    const scene  = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(32, 1, .01, 100);

    /* ---- ENVIRONMENT — metal needs something to reflect ---- */
    if (RoomEnvironment) {
      const pmrem = new THREE.PMREMGenerator(renderer);
      pmrem.compileEquirectangularShader?.();
      scene.environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;
    }

    // gentle fill on top of the environment, not instead of it
    scene.add(new THREE.HemisphereLight(0xffffff, 0x202028, 0.55));
    const kl = new THREE.DirectionalLight(0xffffff, 1.15); kl.position.set(3,5,4);  scene.add(kl);
    const fl = new THREE.DirectionalLight(0xffffff, 0.35); fl.position.set(-3,1,4);  scene.add(fl);
    const rl = new THREE.DirectionalLight(0xc68bff, 0.45); rl.position.set(-4,2,-3); scene.add(rl);

    const pivot = new THREE.Group(); scene.add(pivot);

    const size = () => {
      const r = host.getBoundingClientRect();
      renderer.setSize(r.width, r.height, false);
      camera.aspect = r.width / Math.max(r.height, 1);
      camera.updateProjectionMatrix();
    };
    size(); addEventListener('resize', size);

    let ready = false;
    new GLTFLoader().load(src, (gltf) => {
      const model = gltf.scene;

      // make sure every colour map is tagged sRGB, or textures render washed out
      model.traverse((o) => {
        if (!o.isMesh || !o.material) return;
        const mats = Array.isArray(o.material) ? o.material : [o.material];
        mats.forEach((m) => {
          if (m.map && THREE.SRGBColorSpace) m.map.colorSpace = THREE.SRGBColorSpace;
          if (m.emissiveMap && THREE.SRGBColorSpace) m.emissiveMap.colorSpace = THREE.SRGBColorSpace;
          if (m.envMapIntensity !== undefined) m.envMapIntensity = 1.1;
          m.needsUpdate = true;
        });
      });

      const bb = new THREE.Box3().setFromObject(model);
      const s = new THREE.Vector3(), c = new THREE.Vector3();
      bb.getSize(s); bb.getCenter(c);
      const scale = 1 / Math.max(s.x, s.y, s.z);
      model.scale.setScalar(scale);
      model.position.set(-c.x*scale, -c.y*scale, -c.z*scale);
      pivot.add(model);

      camera.position.set(0, .18, 2.15);
      camera.lookAt(0, 0, 0);

      ready = true;
      fallback.style.opacity = '0';
      setTimeout(() => { fallback.style.display = 'none'; }, 620);
      hint.style.display = 'flex';
    });

    const clamp = v => Math.max(-YAW_LIMIT, Math.min(YAW_LIMIT, v));
    let yaw = 0, down = false, lastX = 0, vel = 0, touched = false, t = 0;

    host.addEventListener('pointerdown', e => {
      down = true; touched = true; vel = 0; lastX = e.clientX;
      host.style.cursor = 'grabbing';
    });
    addEventListener('pointermove', e => {
      if (!down) return;
      vel = (e.clientX - lastX) * .006;
      yaw = clamp(yaw + vel);
      lastX = e.clientX;
    });
    addEventListener('pointerup', () => { down = false; host.style.cursor = 'grab'; });
    host.style.cursor = 'grab';
    host.style.touchAction = 'pan-y';

    (function tick(){
      requestAnimationFrame(tick);
      if (ready) {
        if (down) {
          // yaw set by the move handler
        } else if (!touched) {
          t += 1;
          yaw = Math.sin(t * SWAY_SPEED * 6.28) * (YAW_LIMIT * 0.72);
        } else {
          yaw = clamp(yaw + vel);
          vel *= .92;
          yaw += (0 - yaw) * RETURN_EASE;
        }
        pivot.rotation.y = yaw;
      }
      renderer.render(scene, camera);
    })();
  };

  new IntersectionObserver((entries, obs) => {
    if (entries.some(en => en.isIntersecting)) { obs.disconnect(); boot(); }
  }, { rootMargin: '200px' }).observe(host);
})();
