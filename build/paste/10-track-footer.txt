<script type="module">
import * as THREE from 'three';
import { isLand } from 'js/xero-landmask.js';

const R = 1;
const RIM = R * 1.06;          // outermost thing we must keep in frame
const PAD = 1.12;              // breathing room around it

function latLonToVec(lat, lon, r = R){
  const p = (90 - lat) * Math.PI / 180;
  const t = (lon + 180) * Math.PI / 180;
  return new THREE.Vector3(
    -r * Math.sin(p) * Math.cos(t),
     r * Math.cos(p),
     r * Math.sin(p) * Math.sin(t)
  );
}

/* Rows/step/hex radius are shared by both renderers so the flat version is the same map,
   not a different one that happens to look similar. */
const ROWS  = 118;
const STEP  = 180 / ROWS;
const HEX_R = (Math.PI / ROWS) * 0.60;

/* One WebGL context is the hero's, and it is the one that matters. A phone will happily
   hand out a second, then reclaim the first when it needs the memory back, and the visitor
   scrolls back up to a dead stage. deviceMemory is only reported by Chromium, so the
   viewport width does most of the work here and the memory check is a bonus for the wide
   low-end tablets the breakpoint misses. */
function wantsWebGL(){
  if (window.innerWidth <= 900) return false;
  if (navigator.deviceMemory && navigator.deviceMemory <= 4) return false;
  return true;
}

/* The still frame. Same coastline data, same route, same framing maths - drawn once into
   the 2D context of the same canvas. No GL context, no requestAnimationFrame, so it costs
   the phone one paint and then nothing at all. Redraws on resize only. */
function flatGlobe(host, canvas, origin, dest, progress){
  const ctx = canvas.getContext('2d');
  if (!ctx) return { hexCount: 0 };

  const land = [];
  for (let r = 0; r < ROWS; r++){
    const lat  = 90 - (r + 0.5) * STEP;
    const circ = Math.cos(lat * Math.PI / 180);
    if (circ <= 0.015) continue;
    const cols = Math.max(3, Math.round(ROWS * 2 * circ));
    const off  = (r % 2) ? 0.5 : 0;
    for (let c = 0; c < cols; c++){
      const lon = -180 + ((c + off) / cols) * 360;
      if (isLand(lat, lon)) land.push(latLonToVec(lat, lon, R * 1.004));
    }
  }

  const oV = latLonToVec(origin.lat, origin.lon, R * 1.012);
  const dV = latLonToVec(dest.lat,   dest.lon,   R * 1.012);
  const mid = oV.clone().add(dV).multiplyScalar(.5).normalize()
                .multiplyScalar(R + oV.distanceTo(dV) * .4);
  const arc = [];
  for (let i = 0; i <= 180; i++){
    const t = i / 180, u = 1 - t;
    arc.push(new THREE.Vector3(
      u*u*oV.x + 2*u*t*mid.x + t*t*dV.x,
      u*u*oV.y + 2*u*t*mid.y + t*t*dV.y,
      u*u*oV.z + 2*u*t*mid.z + t*t*dV.z
    ));
  }

  // the settled orientation the animated globe eases to: destination facing the viewer
  const ry = -Math.atan2(dV.x, dV.z), rx = 0.08;
  const cy_ = Math.cos(ry), sy_ = Math.sin(ry), cx_ = Math.cos(rx), sx_ = Math.sin(rx);
  // three applies XYZ euler order, so the group's own rotation is Rx * Ry
  function view(p){
    const x =  p.x * cy_ + p.z * sy_;
    const z = -p.x * sy_ + p.z * cy_;
    return { x: x, y: p.y * cx_ - z * sx_, z: p.y * sx_ + z * cx_ };
  }

  function paint(){
    const b = host.getBoundingClientRect();
    const w = Math.max(b.width, 1), h = Math.max(b.height, 1);
    const dpr = Math.min(devicePixelRatio || 1, 2);
    canvas.width  = Math.round(w * dpr);
    canvas.height = Math.round(h * dpr);
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.clearRect(0, 0, w, h);

    const k = (Math.min(w, h) / 2) / (RIM * PAD);   // same fit rule as the GL camera
    const ox = w / 2, oy = h / 2;
    const sx = p => ox + p.x * k, sy = p => oy - p.y * k;

    ctx.fillStyle = '#14161f';
    ctx.beginPath();
    ctx.arc(ox, oy, R * 0.988 * k, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = '#c3c6d6';
    ctx.beginPath();
    for (let i = 0; i < land.length; i++){
      const p = view(land[i]);
      if (p.z <= 0) continue;                       // far hemisphere
      // foreshorten toward the limb so the disc does not end in a hard ring of full-size cells
      const rr = HEX_R * k * (0.3 + 0.7 * p.z);
      const px = sx(p), py = sy(p);
      for (let a = 0; a < 6; a++){
        const th = a * Math.PI / 3;
        const qx = px + Math.cos(th) * rr, qy = py + Math.sin(th) * rr;
        if (a === 0) ctx.moveTo(qx, qy); else ctx.lineTo(qx, qy);
      }
      ctx.closePath();
    }
    ctx.fill();

    const stroke = (from, to, style, width) => {
      ctx.strokeStyle = style; ctx.lineWidth = width; ctx.lineJoin = 'round';
      ctx.beginPath();
      let pen = false;
      for (let i = from; i <= to; i++){
        const p = view(arc[i]);
        if (p.z <= 0){ pen = false; continue; }     // behind the globe
        const px = sx(p), py = sy(p);
        if (pen) ctx.lineTo(px, py); else { ctx.moveTo(px, py); pen = true; }
      }
      ctx.stroke();
    };
    stroke(0, 180, 'rgba(198,139,255,.18)', 1.4);
    stroke(0, Math.round(180 * progress), 'rgba(198,139,255,.95)', 1.8);

    const dot = (v, style, rad) => {
      const p = view(v);
      if (p.z <= 0) return;
      ctx.fillStyle = style;
      ctx.beginPath(); ctx.arc(sx(p), sy(p), rad * k, 0, Math.PI * 2); ctx.fill();
    };
    const dp = view(dV);
    if (dp.z > 0){
      ctx.strokeStyle = 'rgba(255,122,224,.55)'; ctx.lineWidth = .012 * k;
      ctx.beginPath(); ctx.arc(sx(dp), sy(dp), .034 * k, 0, Math.PI * 2); ctx.stroke();
    }
    dot(oV, '#ffffff', .015);
    dot(dV, '#ff7ae0', .022);
  }

  paint();
  addEventListener('resize', paint);
  if (window.ResizeObserver) new ResizeObserver(paint).observe(host);
  return { hexCount: land.length, flat: true };
}

window.XeroGlobe = function(host, canvas, opts){
  const origin   = opts.origin;
  const dest     = opts.dest;
  const progress = Math.max(.02, Math.min(1, opts.progress == null ? .5 : opts.progress));

  if (!wantsWebGL()) return flatGlobe(host, canvas, origin, dest, progress);

  const renderer = new THREE.WebGLRenderer({ canvas, antialias:true, alpha:true });
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));

  const scene  = new THREE.Scene();
  const FOV = 30;
  const camera = new THREE.PerspectiveCamera(FOV, 1, .1, 100);

  const world = new THREE.Group();
  scene.add(world);

  /* ocean */
  const ocean = new THREE.Mesh(
    new THREE.SphereGeometry(R * 0.988, 128, 128),
    new THREE.MeshBasicMaterial({ color: 0x14161f })
  );
  ocean.renderOrder = 0;
  world.add(ocean);

  /* hexagon landmass (ROWS / STEP / HEX_R are module-level now - the flat renderer
     above builds the same grid from them) */
  const geo  = new THREE.CircleGeometry(HEX_R, 6);
  const mats = [];
  const m = new THREE.Matrix4(), q = new THREE.Quaternion();
  const one = new THREE.Vector3(1,1,1), front = new THREE.Vector3(0,0,1);

  for (let r = 0; r < ROWS; r++){
    const lat  = 90 - (r + 0.5) * STEP;
    const circ = Math.cos(lat * Math.PI / 180);
    if (circ <= 0.015) continue;
    const cols = Math.max(3, Math.round(ROWS * 2 * circ));
    const off  = (r % 2) ? 0.5 : 0;
    for (let c = 0; c < cols; c++){
      const lon = -180 + ((c + off) / cols) * 360;
      if (!isLand(lat, lon)) continue;
      const pos = latLonToVec(lat, lon, R * 1.004);
      q.setFromUnitVectors(front, pos.clone().normalize());
      m.compose(pos, q, one);
      mats.push(m.clone());
    }
  }

  const hexes = new THREE.InstancedMesh(
    geo,
    new THREE.MeshBasicMaterial({ color: 0xc3c6d6, side: THREE.DoubleSide }),
    Math.max(mats.length, 1)
  );
  mats.forEach((mm, i) => hexes.setMatrixAt(i, mm));
  hexes.instanceMatrix.needsUpdate = true;
  hexes.frustumCulled = false;
  hexes.renderOrder = 1;
  world.add(hexes);

  /* route */
  const oV = latLonToVec(origin.lat, origin.lon, R * 1.012);
  const dV = latLonToVec(dest.lat,   dest.lon,   R * 1.012);

  const mk = (v, col, size) => {
    const s = new THREE.Mesh(new THREE.SphereGeometry(size, 20, 20), new THREE.MeshBasicMaterial({ color: col }));
    s.position.copy(v); s.renderOrder = 3; world.add(s); return s;
  };
  mk(oV, 0xffffff, .015);
  const destDot = mk(dV, 0xff7ae0, .022);

  const halo = new THREE.Mesh(
    new THREE.RingGeometry(.028, .04, 48),
    new THREE.MeshBasicMaterial({ color:0xff7ae0, transparent:true, opacity:.85, side:THREE.DoubleSide, depthWrite:false })
  );
  halo.position.copy(dV);
  halo.lookAt(dV.clone().multiplyScalar(2));
  halo.renderOrder = 3;
  world.add(halo);

  const mid   = oV.clone().add(dV).multiplyScalar(.5).normalize()
                  .multiplyScalar(R + oV.distanceTo(dV) * .4);
  const curve = new THREE.QuadraticBezierCurve3(oV, mid, dV);
  const full  = curve.getPoints(180);

  const faint = new THREE.Line(new THREE.BufferGeometry().setFromPoints(full),
    new THREE.LineBasicMaterial({ color:0xc68bff, transparent:true, opacity:.18 }));
  faint.renderOrder = 2; world.add(faint);

  const lit = new THREE.Line(
    new THREE.BufferGeometry().setFromPoints(full.slice(0, Math.round(180 * progress) + 1)),
    new THREE.LineBasicMaterial({ color:0xc68bff, transparent:true, opacity:.95 }));
  lit.renderOrder = 2; world.add(lit);

  const pulse = new THREE.Mesh(new THREE.SphereGeometry(.019, 14, 14),
    new THREE.MeshBasicMaterial({ color:0xffffff }));
  pulse.renderOrder = 3; world.add(pulse);

  /* framing — solve the distance so nothing can crop */
  const targetY = -Math.atan2(dV.x, dV.z);
  world.rotation.y = targetY - 1.15;
  world.rotation.x = 0.08;
  let spinTarget = targetY;

  function fit(){
    const b = host.getBoundingClientRect();
    const w = Math.max(b.width, 1), h = Math.max(b.height, 1);
    renderer.setSize(w, h, false);
    const aspect = w / h;
    camera.aspect = aspect;

    const vFOV = FOV * Math.PI / 180;
    const distV = RIM / Math.tan(vFOV / 2);          // fit vertically
    const hFOV  = 2 * Math.atan(Math.tan(vFOV / 2) * aspect);
    const distH = RIM / Math.tan(hFOV / 2);          // fit horizontally
    camera.position.set(0, 0, Math.max(distV, distH) * PAD);
    camera.lookAt(0, 0, 0);
    camera.updateProjectionMatrix();
  }
  fit();
  addEventListener('resize', fit);
  if (window.ResizeObserver) new ResizeObserver(fit).observe(host);

  let down = false, lastX = 0, vel = 0;
  host.addEventListener('pointerdown', e => { down = true; lastX = e.clientX; spinTarget = null; });
  addEventListener('pointermove', e => {
    if (!down) return;
    vel = (e.clientX - lastX) * .006;
    world.rotation.y += vel;
    lastX = e.clientX;
  });
  addEventListener('pointerup', () => { down = false; });
  host.style.cursor = 'grab';
  host.style.touchAction = 'pan-y';

  let t = 0;
  (function tick(){
    requestAnimationFrame(tick);
    t += .0035;
    const k = (t % 1) * progress;
    pulse.position.copy(curve.getPoint(k));
    pulse.scale.setScalar(.7 + Math.sin((t % 1) * Math.PI) * .9);
    halo.scale.setScalar(1 + (t * 2 % 1) * 1.7);
    halo.material.opacity = .85 * (1 - (t * 2 % 1));
    destDot.scale.setScalar(1 + Math.sin(t * 6) * .12);
    if (spinTarget !== null && !down){
      world.rotation.y += (spinTarget - world.rotation.y) * .02;
    } else if (!down){
      world.rotation.y += vel; vel *= .95; world.rotation.y += .0005;
    }
    renderer.render(scene, camera);
  })();

  return { hexCount: mats.length };
};
</script>
